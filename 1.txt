1. List title, rating and length for all films, longest first.

QUERY:
select f.title, f.length, f.rating 
from film f 
order by f.length desc;



2.List the distinct ratings.
QUERY:
select DISTINCT(f.rating)
from film f;


3.Find films with rental_rate > 2.99 and length < 90.
QUERY:
select * 
from film f 
where f.rental_rate < 2.99 and f.length < 90;

4.Count customers per store.
QUERY:
select COUNT(customer), store_id as store_name
from customer
group by store_id;

5.Find active customers whose last name starts with “S”.
QUERY:
select * from customer c 
where c.first_name like 'S%';

6.Find films with “love” anywhere in the title, case-insensitive.
QUERY:
select * from film f 
where f.title like '%love%';

7.Count films per rating, keeping only ratings with more than 190 films (HAVING).
QUERY:
select f.title as films_name, f.rating 
from film f
group by f.film_id  
having COUNT(f.rating ) < 190;

8.Find the average rental_duration per rating.
QUERY:
select AVG(f.rental_duration) as average_rental_duration, f.rating
from film f 
group by f.rating;
